-- Customer Lifetime Value Analysis with Cohorts
WITH customer_cohorts AS (
    SELECT 
        customer_id,
        DATE_TRUNC('month', first_order_date) AS cohort_month,
        EXTRACT(YEAR FROM AGE(order_date, first_order_date)) * 12 + 
        EXTRACT(MONTH FROM AGE(order_date, first_order_date)) AS period_number
    FROM (
        SELECT 
            customer_id,
            order_date,
            MIN(order_date) OVER (PARTITION BY customer_id) AS first_order_date
        FROM orders
        WHERE status = 'completed'
    ) customer_orders
),
cohort_sizes AS (
    SELECT 
        cohort_month,
        COUNT(DISTINCT customer_id) AS cohort_size
    FROM customer_cohorts 
    WHERE period_number = 0
    GROUP BY cohort_month
),
retention_table AS (
    SELECT 
        c.cohort_month,
        c.period_number,
        COUNT(DISTINCT c.customer_id) AS active_customers,
        cs.cohort_size
    FROM customer_cohorts c
    JOIN cohort_sizes cs ON c.cohort_month = cs.cohort_month
    GROUP BY c.cohort_month, c.period_number, cs.cohort_size
)
SELECT 
    cohort_month,
    period_number,
    active_customers,
    cohort_size,
    ROUND(100.0 * active_customers / cohort_size, 2) AS retention_rate
FROM retention_table
ORDER BY cohort_month, period_number;

-- Advanced Statistical Analysis with Window Functions
WITH daily_metrics AS (
    SELECT 
        date_trunc('day', created_at) AS metric_date,
        COUNT(*) AS daily_orders,
        SUM(total_amount) AS daily_revenue,
        AVG(total_amount) AS avg_order_value
    FROM orders
    WHERE created_at >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY date_trunc('day', created_at)
),
statistical_analysis AS (
    SELECT 
        metric_date,
        daily_orders,
        daily_revenue,
        avg_order_value,
        -- Moving averages
        AVG(daily_orders) OVER (
            ORDER BY metric_date 
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ) AS orders_7day_ma,
        AVG(daily_revenue) OVER (
            ORDER BY metric_date 
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ) AS revenue_7day_ma,
        -- Statistical measures
        STDDEV(daily_orders) OVER (
            ORDER BY metric_date 
            ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
        ) AS orders_30day_stddev,
        -- Anomaly detection using z-score
        (daily_orders - AVG(daily_orders) OVER (
            ORDER BY metric_date 
            ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
        )) / NULLIF(STDDEV(daily_orders) OVER (
            ORDER BY metric_date 
            ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
        ), 0) AS orders_zscore
    FROM daily_metrics
)
SELECT 
    *,
    CASE 
        WHEN ABS(orders_zscore) > 2 THEN 'Anomaly'
        WHEN ABS(orders_zscore) > 1.5 THEN 'Outlier'
        ELSE 'Normal'
    END AS orders_classification
FROM statistical_analysis
ORDER BY metric_date;

### Python Integration via API
```python
# Python SDK for QuartzIQ Integration
import quartziq
from sklearn.ensemble import RandomForestRegressor
import pandas as pd

# Initialize QuartzIQ client
client = quartziq.Client(
    api_key='your-api-key',
    base_url='https://api.quartziq.com'
)

# Fetch data for model training
query = """
    SELECT 
        customer_id,
        days_since_last_order,
        total_orders,
        avg_order_value,
        total_spent,
        will_churn_next_30_days
    FROM customer_features
    WHERE created_at >= CURRENT_DATE - INTERVAL '2 years'
"""

df = client.execute_query(query).to_dataframe()

# Train custom model
features = ['days_since_last_order', 'total_orders', 'avg_order_value', 'total_spent']
X = df[features]
y = df['will_churn_next_30_days']

model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(X, y)

# Generate predictions
current_customers = client.execute_query("""
    SELECT customer_id, days_since_last_order, total_orders, 
           avg_order_value, total_spent
    FROM current_customer_features
""").to_dataframe()

predictions = model.predict(current_customers[features])
current_customers['churn_probability'] = predictions

# Upload predictions back to QuartzIQ
client.upload_dataframe(
    current_customers,
    table_name='customer_churn_predictions',
    mode='replace'
)
```

const QuartzIQCustomViz = {
  // Advanced network graph visualization
  createNetworkGraph: function(data, container, options) {
    const width = options.width || 800;
    const height = options.height || 600;
    
    const svg = d3.select(container)
      .append("svg")
      .attr("width", width)
      .attr("height", height);
    
    const simulation = d3.forceSimulation(data.nodes)
      .force("link", d3.forceLink(data.links).id(d => d.id))
      .force("charge", d3.forceManyBody().strength(-300))
      .force("center", d3.forceCenter(width / 2, height / 2));
    
    const link = svg.append("g")
      .selectAll("line")
      .data(data.links)
      .enter().append("line")
      .attr("stroke", "#999")
      .attr("stroke-width", d => Math.sqrt(d.value));
    
    const node = svg.append("g")
      .selectAll("circle")
      .data(data.nodes)
      .enter().append("circle")
      .attr("r", d => d.size || 5)
      .attr("fill", d => d.color || "#69b3a2")
      .call(d3.drag()
        .on("start", dragstarted)
        .on("drag", dragged)
        .on("end", dragended));
    
    simulation.on("tick", () => {
      link
        .attr("x1", d => d.source.x)
        .attr("y1", d => d.source.y)
        .attr("x2", d => d.target.x)
        .attr("y2", d => d.target.y);

        .attr("cx", d => d.x)
        .attr("cy", d => d.y);

      d.fy = d.y;

    function dragged(event, d) {
      d.fx = event.x;

      if (!event.active) simulation.alphaTarget(0);
      d.fx = null;
      d.fy = null;
    }
  },

# Troubleshooting Authentication Problems

## Single Sign-On (SSO) Issues
**Symptoms**: Unable to login via corporate SSO
**Common Causes**:
1. SAML configuration mismatch
2. Certificate expiration  
3. Attribute mapping errors
4. Network connectivity issues

**Resolution Steps**:
1. Verify SAML metadata configuration
2. Check certificate validity dates
3. Validate attribute mappings (email, role, department)
4. Test connectivity to identity provider
5. Review audit logs for error messages

**Escalation**: If issue persists after basic checks, escalate to Tier 2 with:
- User email and organization
- Error messages from browser console
- Network trace logs if available
- Identity provider configuration details

## Multi-Factor Authentication (MFA) Problems
**Symptoms**: MFA codes not working or not received
**Resolution Steps**:
1. Verify device time synchronization
2. Check spam/junk folders for SMS/email codes
3. Try alternative MFA method if configured
4. Reset MFA device registration if necessary
5. Contact administrator for temporary MFA bypass
```

#### Performance and Loading Issues
```markdown
# Dashboard Performance Troubleshooting

## Slow Dashboard Loading
**Symptoms**: Dashboards taking >10 seconds to load
**Diagnostic Steps**:
1. Check browser developer tools for network timing
2. Identify slow-loading widgets or data sources
3. Review query complexity and data volume
4. Check for concurrent user load

**Optimization Techniques**:
```sql
-- Query optimization example
-- Before: Unoptimized query
SELECT * FROM large_table 
WHERE date_column >= '2024-01-01'
ORDER BY date_column DESC;

-- After: Optimized with proper indexing and limits
SELECT 
    essential_columns_only
FROM large_table 
WHERE date_column >= CURRENT_DATE - INTERVAL '90 days'
    AND status = 'active'  -- Additional selective filter
ORDER BY date_column DESC
LIMIT 1000;  -- Reasonable limit for dashboard display
```

**Caching Strategies**:
1. Enable automatic query result caching
2. Schedule dashboard refreshes during off-peak hours
3. Use summary tables for historical analysis
4. Implement progressive loading for large datasets

## Memory and Resource Issues
**Symptoms**: Browser crashes, system slowdown
**Resolution Steps**:
1. Close unnecessary browser tabs
2. Clear browser cache and cookies
3. Restart browser session
4. Check system memory availability
5. Use dashboard pagination for large datasets

**Prevention**:
- Implement data limits on dashboard queries
- Use incremental loading for time-series data
- Optimize visualizations for performance
- Monitor resource usage patterns
```

**Symptoms**: API calls failing or returning errors

1. Verify API endpoint URLs and authentication

3. Validate request format and parameters
4. Review API response codes and error messages
5. Test API calls outside QuartzIQ environment

**Common API Response Codes**:
- 401: Authentication failure - check API keys
- 403: Authorization denied - verify permissions  
- 429: Rate limit exceeded - implement retry logic
- 500: Server error - contact API provider support
